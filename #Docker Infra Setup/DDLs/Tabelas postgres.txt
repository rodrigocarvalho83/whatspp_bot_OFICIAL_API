-- observability.bot_events definição

-- Drop table

-- DROP TABLE observability.bot_events;

CREATE TABLE observability.bot_events (
	id bigserial NOT NULL,
	received_at timestamptz DEFAULT now() NOT NULL,
	event_timestamp timestamptz NULL,
	service text NOT NULL,
	environment text NOT NULL,
	"event" text NOT NULL,
	message text NOT NULL,
	extra jsonb DEFAULT '{}'::jsonb NOT NULL,
	headers jsonb DEFAULT '{}'::jsonb NOT NULL,
	query_params jsonb DEFAULT '{}'::jsonb NOT NULL,
	route_params jsonb DEFAULT '{}'::jsonb NOT NULL,
	raw_body jsonb DEFAULT '{}'::jsonb NOT NULL,
	webhook_url text NULL,
	execution_mode text NULL,
	source_ip text NULL,
	CONSTRAINT bot_events_pkey PRIMARY KEY (id)
);
CREATE INDEX idx_bot_events_event ON observability.bot_events USING btree (event);
CREATE INDEX idx_bot_events_event_ts ON observability.bot_events USING btree (event_timestamp DESC);
CREATE INDEX idx_bot_events_extra_gin ON observability.bot_events USING gin (extra);
CREATE INDEX idx_bot_events_received_at ON observability.bot_events USING btree (received_at DESC);
CREATE INDEX idx_bot_events_service_env ON observability.bot_events USING btree (service, environment);
